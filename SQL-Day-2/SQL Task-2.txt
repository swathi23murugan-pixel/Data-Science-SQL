-- MOST BORROWED BOOK
SELECT Book.book_name,COUNT(Borrow.borrow_id) AS borrow_count FROM Book
JOIN Borrow ON Book.book_id=Borrow.book_id
GROUP BY Book.book_id,Book.book_name
ORDER BY borrow_count DESC LIMIT 1;

JOIN Borrow ON Members.member_id=Borrow.member_id
GROUP BY Members.member_id,Members.member_name
HAVING COUNT(Borrow.borrow_id)>2;

SELECT Book.* FROM Book,avg_price WHERE Book.price>avg_price.average_price;

WITH category_price AS(SELECT category,AVG(price) AS average_price FROM Book GROUP BY category)
SELECT * FROM category_price;

WITH category_price AS(SELECT category,AVG(price) AS average_price FROM Book GROUP BY category)
SELECT Book.* FROM Book
JOIN category_price ON Book.category=category_price.category
WHERE Book.price>category_price.average_price;

SELECT *,MAX(price) OVER(PARTITION BY category) AS highest_price FROM Book;

SELECT *,MIN(price) OVER(PARTITION BY category) AS lowest_price FROM Book;
