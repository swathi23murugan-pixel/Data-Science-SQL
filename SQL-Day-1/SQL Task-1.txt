-- DATABASE

USE librarydb;


-- TABLE CREATION

CREATE TABLE Book(
book_id INT PRIMARY KEY,
book_name VARCHAR(50),
author VARCHAR(100),
price DECIMAL(10,4)
);


-- MEMBER TABLE

CREATE TABLE Members(
member_id INT PRIMARY KEY,
member_name VARCHAR(50),
city VARCHAR(100),
phone VARCHAR(15)
);


-- BORROW TABLE

CREATE TABLE Borrow(
borrow_id INT PRIMARY KEY,
book_id INT,
member_id INT,
borrow_date DATE
);


-- ADD COLUMNS

ALTER TABLE Book ADD category VARCHAR(100);
ALTER TABLE Book ADD stock_quantity INT;
ALTER TABLE Members ADD email VARCHAR(100);
